cmake_minimum_required(VERSION 3.22)
project(voicetxt_hunyuan LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama-build)
target_compile_definitions(ggml-base PRIVATE VOICETXT_HYMT_STQ42=1)
add_library(hunyuan-jni SHARED hunyuan-jni.cpp)
target_include_directories(hunyuan-jni PRIVATE "${LLAMA_SOURCE}/include")
target_link_libraries(hunyuan-jni PRIVATE llama android log)
target_link_options(hunyuan-jni PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
add_executable(hy-stq-check EXCLUDE_FROM_ALL stq-check.cpp)
target_include_directories(hy-stq-check PRIVATE "${LLAMA_SOURCE}/ggml/src")
target_link_libraries(hy-stq-check PRIVATE ggml ggml-cpu android log)
add_executable(hy-inference-check EXCLUDE_FROM_ALL hunyuan-check.cpp)
target_include_directories(hy-inference-check PRIVATE "${LLAMA_SOURCE}/include")
target_link_libraries(hy-inference-check PRIVATE llama android log)
